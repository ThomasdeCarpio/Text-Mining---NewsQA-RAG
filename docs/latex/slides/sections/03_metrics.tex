\rowcolor{paleblue}\textbf{Thước đo} & \textbf{Nội dung đo} \\\midrule

Recall@5 & Tìm được bao nhiêu phần gold chunks? \\\midrule
MRR@5 & Gold chunk đầu tiên đứng cao đến đâu? \\\midrule

\rowcolor{palegreen}\textbf{Thước đo} & \textbf{Nội dung đo} \\\midrule
Answer Correctness & Nội dung có khớp đáp án chuẩn? \\\midrule
Exact Match & Khớp hoàn toàn sau chuẩn hóa? \\\midrule
Token F1 & Token đáp án khớp đúng và đủ? \\\midrule
Faithfulness & Khẳng định có được context hỗ trợ? \\\midrule
Citation F1 & Trích đúng và đủ gold chunks? \\\midrule
Citation Validity & Số citation có tồn tại trong context? \\\bottomrule
\end{tabularx}}

\takeaway{\textbf{Phase 1:} ưu tiên MRR@5, đối chiếu nDCG@5 và Hit@5.\quad
\textbf{Phase 2:} ưu tiên Answer Correctness; Faithfulness và citation là guardrail.}
\slidenote{Gold chunk: đoạn chứa evidence chuẩn. Context: các đoạn đưa cho generator. Mọi điểm chất lượng nằm trong khoảng 0–1; cao hơn tốt hơn.}
