\heading{Thiết lập và phân tích đánh đổi}
{\fontsize{13}{16.5}\selectfont
\begin{itemize}
  \setlength{\itemsep}{1.5pt}
  \item \textbf{Thiết lập:} Thêm 1 ví dụ tổng hợp (Northbridge/Riverside, không thuộc corpus) vào prompt P2; giữ cùng truy xuất, depth 5, Gemini 3.1 Flash-Lite.
  \item \textbf{Cải thiện mạnh:} AC tăng $+0{,}0664$, Token F1 tăng $+0{,}2076$; Exact Match tăng gần $4\times$ (từ $0{,}0925 \rightarrow 0{,}3559$) nhờ mô hình bắt chước trả lời ngắn đúng loại thực thể.
  \item \textbf{Đánh đổi:} Faithfulness giảm nhẹ ($-0{,}0199$, sát ngưỡng guardrail $0{,}02$). Relevancy giảm mạnh do đáp án quá ngắn (trung vị giảm $11 \rightarrow 7$ từ) làm judge sinh câu hỏi ngược thiếu chủ thể.
\end{itemize}\par}

\takeaway{One-shot tăng mạnh độ đúng và Token F1 nhưng đánh đổi Faithfulness và Relevancy. Khảo sát giải thích đánh đổi, không thay cấu hình zero-shot đã chốt.}
\slidenote{Đánh giá trên 281 câu development; đối chứng P2 zero-shot được tái sử dụng. Trên held-out, P2-1S đạt AC = 0,7557.}
