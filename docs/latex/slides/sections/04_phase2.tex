\begin{frame}[shrink=20]{\texttt{context\_depth} là gì}{Thứ được tinh chỉnh song song với prompt}
  \small
  Đúng một phép cắt danh sách trên vết đã xếp hạng sẵn:
  \begin{center}
    \texttt{generation\_chunks = ranked\_chunks[:context\_depth]}
  \end{center}
  \textbf{Không truy xuất lại, không xếp hạng lại} --- nên chênh lệch giữa các
  mức depth quy hết được cho tầng sinh.

  \vspace{0.5em}
  Model nhận được khối context đánh số, và chính các số này là số nó trích dẫn:
  \begin{center}\footnotesize
    \texttt{Context:}\quad\texttt{[1] \ldots}\quad\texttt{[2] \ldots}\quad\texttt{[3] \ldots}\\
    \texttt{Question: \ldots}\qquad\texttt{Answer:}
  \end{center}

  \begin{tabularx}{\textwidth}{@{}p{1.2cm} p{2.6cm} L@{}}
    \toprule
    \textbf{depth} & \textbf{Model thấy} & \textbf{Đánh đổi} \\
    \midrule
    1 & \texttt{[1]} & Ít token nhất --- nhưng \fail{\NumLostAtDepthOne{}/\NumNDev} câu mất sạch bằng chứng \\
    3 & \texttt{[1] [2] [3]} & Cân bằng --- nhưng \fail{\NumLostAtDepthThree{}/\NumNDev} câu mất sạch bằng chứng \\
    5 & \texttt{[1]}\ldots\texttt{[5]} & \pass{Không mất bằng chứng} --- tốn token nhất \\
    \bottomrule
  \end{tabularx}

  \takeaway{Cột bên phải \emph{không} đến từ ước lượng --- Phase 1 đã đo sẵn nó.}
\end{frame}

\begin{frame}[shrink=20]{Quyết định winner --- guardrail chạy trước, điểm số chạy sau}
  \small
  \textbf{Bốn điều kiện khóa trước khi nhìn thấy bất kỳ kết quả nào:}
  Faithfulness $\geq -0{,}02$ \;\textbullet\; Citation F1 $\geq -0{,}01$
  \;\textbullet\; Citation Validity $\geq -0{,}01$ \;\textbullet\; coverage $\geq 95\%$.
  \emph{Chỉ khi qua cả bốn} mới xét Answer Correctness cao nhất.

  \vspace{0.5em}
  \begin{tabularx}{\textwidth}{@{}L N N N@{}}
    \toprule
    & \textbf{P0} & \textbf{P2-depth3} & \textbf{P2-depth5} \\
    \midrule
    Answer Correctness & \NumPzeroAC & \emph{\NumPtwodThreeAC} & \win{\NumPtwodFiveAC} \\
    Exact Match & \NumPzeroEM & \emph{\NumPtwodThreeDeltaEM} & \NumPtwodFiveDeltaEM \\
    \midrule
    Faithfulness $\Delta$ & --- & \fail{\NumPtwodThreeGuardFaith} & \pass{\NumPtwodFiveGuardFaith} \\
    Citation F1 $\Delta$ & --- & \pass{\NumPtwodThreeGuardCitFone} & \pass{\NumPtwodFiveGuardCitFone} \\
    Citation Validity $\Delta$ & --- & \fail{\NumPtwodThreeGuardCitVal} & \pass{\NumPtwodFiveGuardCitVal} \\
    \midrule
    \textbf{Phán quyết} & --- & \fail{LOẠI} & \win{CHỌN} \\
    \bottomrule
  \end{tabularx}

  \vspace{0.4em}
  P2-depth3 \textbf{điểm cao hơn} nhưng trượt 2 trong 4 guardrail. Kiểm định
  theo cặp xác nhận \textbf{cả hai mức tụt đều là thật}
  (Faithfulness \NumPtwodThreeDeltaFaith{} \muted{\NumPtwodThreeDeltaFaithCI}, CI95 không chứa 0).

  \takeaway{Nới ngưỡng lúc này chính là hành vi mà việc đăng ký trước sinh ra để ngăn.}
\end{frame}
